//845. Longest Mountain in Array
class Solution {
    public int longestMountain(int[] arr) {
        int start=0,end=0;
        int n=arr.length;
        int maxc=0;
        for(int i=1;i<n-1;i++){
            int right=0,left=0;
            int rc=i-1,fc=i+1,rcv=arr[i],fcv=arr[i];
            while(rc>=0){
                if(arr[rc]<rcv){
                    right++;
                    rcv=arr[rc];
                    rc--;
                }
                else{
                    break;
                }
            }
            while(fc<=n-1){
                if(arr[fc]<fcv){
                    left++;
                    fcv=arr[fc];
                    fc++;
                }
                else{
                    break;
                }
            }
            if(right!=0 && left!=0 && maxc<left+right) maxc=left+right;
        }
        if(maxc==0) return 0;
        return maxc+1;
    }
}
