//724. Find Pivot Index
class Solution {
    public int pivotIndex(int[] nums) {
        int sum=0;
        int temp=0;
        for(int i:nums){
            sum+=i;
        }
        for(int i=0;i<nums.length;i++){
            temp+=nums[i];
            if(temp==sum){
                return i;
            }
            sum-=nums[i];
        }
        return -1;
    }
}
