import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.nextLine();

        String res = "";
        int i = s.length() - 1;

        while(i >= 0) {
            while(i >= 0 && s.charAt(i) == ' ') i--;

            int end = i;

            while(i >= 0 && s.charAt(i) != ' ') i--;

            int start = i + 1;

            for(int j = start; j <= end; j++) {
                res = res + s.charAt(j);
            }

            res = res + " ";
        }

        System.out.print(res.trim());
    }
}
